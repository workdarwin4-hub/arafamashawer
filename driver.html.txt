    <h1>لوحة تحكم المندوب - شبرا الخيمة</h1>
    <hr>
    <h2>الطلبات المتاحة في الزون:</h2>
    <div id="ordersList">جاري تحميل الطلبات...</div>
</div>

<script type="module">
    import { db } from './js/firebase-config.js';
    import { collection, getDocs } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";

    async function loadOrders() {
        const listDiv = document.getElementById('ordersList');
        listDiv.innerHTML = "";
        
        try {
            const querySnapshot = await getDocs(collection(db, "orders"));
            if (querySnapshot.empty) {
                listDiv.innerHTML = "<p>لا توجد طلبات جديدة حالياً.</p>";
                return;
            }

            querySnapshot.forEach((doc) => {
                const order = doc.data();
                const card = document.createElement('div');
                card.className = "order-card";
                card.innerHTML = `
                    <h3>العميل: ${order.clientName}</h3>
                    <p><strong>رقم الهاتف:</strong> ${order.clientPhone}</p>
                    <p><strong>العنوان:</strong> ${order.clientAddress}</p>
                    <p><strong>الطلب:</strong> ${order.orderDetails}</p>
                    <p><strong>الحالة:</strong> ${order.status}</p>
                    <button class="btn-accept">قبول الطلب</button>
                `;
                listDiv.appendChild(card);
            });
        } catch (error) {
            listDiv.innerHTML = "<p style='color:red;'>حدث خطأ أثناء تحميل الطلبات.</p>";
            console.error(error);
        }
    }

    loadOrders();
</script>
