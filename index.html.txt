        </div>

            <textarea id="orderDetails" rows="4" required placeholder="اكتب الأطلبية بالتفصيل..."></textarea>
        </div>
        <button type="submit" id="submitBtn">إرسال الطلب الآن</button>

<script type="module">
    import { db } from './js/firebase-config.js';
    import { collection, addDoc } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";

    const form = document.getElementById('orderForm');
    const alertDiv = document.getElementById('statusAlert');

    form.addEventListener('submit', async (e) => {
        e.preventDefault();
        
        const name = document.getElementById('clientName').value;
        const phone = document.getElementById('clientPhone').value;
        const address = document.getElementById('clientAddress').value;
        const details = document.getElementById('orderDetails').value;

        try {
            await addDoc(collection(db, "orders"), {
                clientName: name,
                clientPhone: phone,
                clientAddress: address,
                orderDetails: details,
                status: "جديد",
                createdAt: new Date()
            });

            alertDiv.className = "alert success";
            alertDiv.innerText = "تم إرسال طلبك بنجاح وسيتواصل معك المندوب قريباً!";
            form.reset();
        } catch (error) {
            alertDiv.className = "alert danger";
            alertDiv.innerText = "حدث خطأ أثناء إرسال الطلب، تأكد من الاتصال بالإنترنت.";
            console.error("Error adding document: ", error);
        }
    });
</script>
